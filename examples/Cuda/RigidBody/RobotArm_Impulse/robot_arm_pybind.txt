#include <pybind11/pybind11.h>
#include <pybind11/stl.h>
#include <pybind11/eigen.h>
#include "RobotArm.h"
#include "DataTypes.h"

namespace py = pybind11;
using namespace dyno;

// 绑定Vec3f类型
void bindVec3f(py::module& m) {
    py::class_<Vec3f>(m, "Vec3f")
        .def(py::init<float, float, float>(), py::arg("x") = 0, py::arg("y") = 0, py::arg("z") = 0)
        .def_readwrite("x", &Vec3f::x)
        .def_readwrite("y", &Vec3f::y)
        .def_readwrite("z", &Vec3f::z)
        .def("__repr__", [](const Vec3f& v) {
            return "Vec3f(" + std::to_string(v.x) + ", " + std::to_string(v.y) + ", " + std::to_string(v.z) + ")";
        });
}

// 绑定Quat类型
void bindTQuat(py::module& m) {
     py::class_<TQuat>(m, "TQuat")
         // C++ 构造函数是 Quat(Real _x, Real _y, Real _z, Real _w)
         // 这里按 (x, y, z, w) 暴露给 Python，默认是单位四元数
         .def(py::init<float, float, float, float>(),
              py::arg("x") = 0.0f,
              py::arg("y") = 0.0f,
              py::arg("z") = 0.0f,
              py::arg("w") = 1.0f)
         .def_readwrite("w", &TQuat::w)
         .def_readwrite("x", &TQuat::x)
         .def_readwrite("y", &TQuat::y)
         .def_readwrite("z", &TQuat::z)
         .def("__repr__", [](const TQuat& q) {
             // 与构造参数顺序保持一致：TQuat(x, y, z, w)
             return "TQuat(" +
                    std::to_string(q.x) + ", " +
                    std::to_string(q.y) + ", " +
                    std::to_string(q.z) + ", " +
                    std::to_string(q.w) + ")";
         });
}

// 绑定RobotArmSimulator类
void bindRobotArmSimulator(py::module& m) {
    py::class_<RobotArmSimulator<DataType3f>>(m, "RobotArmSimulator")
        .def(py::init<>())
        .def("createScene", &RobotArmSimulator<DataType3f>::createScene)
        .def("addRigidSystem", &RobotArmSimulator<DataType3f>::addRigidSystem,
             py::arg("offset"), py::arg("targetPosition"), py::arg("density") = 1000.0f,
             "添加机械臂系统，返回rigidID")
        .def("reset", &RobotArmSimulator<DataType3f>::reset,
             py::arg("rigidID") = -1, py::arg("targetPosition") = Vec3f(0.0f, 0.0f, 0.0f),
             "重置机械臂系统，-1表示重置所有")
        .def("setupSceneGraph", &RobotArmSimulator<DataType3f>::setupSceneGraph)
        .def("initialize", &RobotArmSimulator<DataType3f>::initialize,
             py::arg("width") = 1280, py::arg("height") = 768, py::arg("scale") = 1.0f,
             "初始化仿真环境")
        .def("stepSimulation", &RobotArmSimulator<DataType3f>::stepSimulation,
             py::arg("moterVelocities"), py::arg("enableRendering") = true,
             "执行单步仿真")
        .def("terminateSimulation", &RobotArmSimulator<DataType3f>::terminateSimulation,
             "终止仿真")
        .def("setMoters", &RobotArmSimulator<DataType3f>::setMoters,
             py::arg("moterImpulses"))
        .def("getRigidSystemCount", &RobotArmSimulator<DataType3f>::getRigidSystemCount,
             "获取机械臂系统数量")
        .def("fingerPosition", &RobotArmSimulator<DataType3f>::fingerPosition,
             py::arg("rigidID"), "获取指定机械臂的手指位置")
        .def("rigidPosition", &RobotArmSimulator<DataType3f>::rigidPosition,
             py::arg("systemID"), py::arg("rigidID"), "获取指定刚体位置")
        .def("rigidRotation", &RobotArmSimulator<DataType3f>::rigidRotation,
             py::arg("systemID"), py::arg("rigidID"), "获取指定刚体旋转四元数")
        .def("rigidVelocity", &RobotArmSimulator<DataType3f>::rigidVelocity,
             py::arg("systemID"), py::arg("rigidID"), "获取指定刚体线速度")
        .def("rigidAngularVelocity", &RobotArmSimulator<DataType3f>::rigidAngularVelocity,
             py::arg("systemID"), py::arg("rigidID"), "获取指定刚体角速度")
        .def_property("target_center", 
             [](RobotArmSimulator<DataType3f>& self) { return self.varTargetCenter()->getValue(); },
             [](RobotArmSimulator<DataType3f>& self, const Vec3f& val) { self.varTargetCenter()->setValue(val); },
             "目标位置属性");
}

// 模块定义
PYBIND11_MODULE(RobotArm, m) {
    m.doc() = "RobotArm simulator Python bindings";
    
    // 绑定向量类型
    bindVec3f(m);
    
    // 绑定仿真器类
    bindRobotArmSimulator(m);
}
